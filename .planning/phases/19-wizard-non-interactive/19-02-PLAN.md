Create input reader that handles both file paths and stdin per CONTEXT.md:

```typescript
import { readFile } from 'node:fs/promises';

export async function readJsonInput(source: string): Promise<unknown> {
  let content: string;

  if (source === '-') {
    // Read from stdin
    content = await readStdin();
  } else {
    // Read from file
    content = await readFile(source, 'utf-8');
  }

  return JSON.parse(content);
}

async function readStdin(): Promise<string> {
  // Per RESEARCH.md pitfall: check TTY before reading to avoid hang
  if (process.stdin.isTTY) {
    throw new Error('Cannot read from stdin in interactive mode. Use a file path instead of "-"');
  }

  const chunks: Buffer[] = [];
  for await (const chunk of process.stdin) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf-8');
}
```

import { z } from 'zod';
import { WizardInitInputSchema, AddExperienceInputSchema, ... } from './schemas.js';

type SchemaCommand = 'init' | 'add-experience' | 'add-skills' | 'add-project' | 'add-certification' | 'add-education';

export function getJsonSchema(command: SchemaCommand): object {
  const schemas: Record<SchemaCommand, z.ZodType> = {
    'init': WizardInitInputSchema,
    'add-experience': AddExperienceInputSchema,
    // ... etc
  };
  return z.toJSONSchema(schemas[command]);
}

export function showJsonSchema(command: SchemaCommand): void {
  console.log(JSON.stringify(getJsonSchema(command), null, 2));
}
