```typescript
import type { WizardState } from '../types.js';

/**
 * CLI flag options for wizard init command.
 * These map to Commander.js options.
 */
export interface InitFlagOptions {
  // Contact flags
  name?: string;
  email?: string;
  phone?: string;
  location?: string;
  // Simple link flags (for quick single-link add)
  linkedIn?: string;
  github?: string;
  website?: string;
}

/**
 * CLI flag options for add experience command.
 */
export interface AddExperienceFlagOptions {
  company?: string;
  role?: string;
  startDate?: string;
  endDate?: string;
  location?: string;
  bullets?: string[];  // Commander supports --bullet "x" --bullet "y"
  techStack?: string;  // Comma-separated
}

/**
 * Build partial wizard state from init flags.
 * Does NOT validate - validation happens at schema level.
 */
export function buildContactFromFlags(flags: InitFlagOptions): Partial<WizardState> {
  if (!flags.name) {
    return {};
  }

  const links: Array<{type: string; url: string; label?: string}> = [];
  if (flags.linkedIn) links.push({ type: 'linkedin', url: normalizeUrl(flags.linkedIn) });
  if (flags.github) links.push({ type: 'github', url: normalizeUrl(flags.github) });
  if (flags.website) links.push({ type: 'website', url: normalizeUrl(flags.website) });

  return {
    contact: {
      name: flags.name,
      ...(flags.email && { email: flags.email }),
      ...(flags.phone && { phone: flags.phone }),
      ...(flags.location && { location: flags.location }),
      ...(links.length > 0 && { links }),
    },
  };
}

/**
 * Build single experience entry from flags.
 */
export function buildExperienceFromFlags(flags: AddExperienceFlagOptions): WorkExperience | null {
  if (!flags.company || !flags.role || !flags.startDate) {
    return null;
  }

  return {
    company: flags.company,
    role: flags.role,
    startDate: flags.startDate,
    endDate: flags.endDate ?? 'present',
    ...(flags.location && { location: flags.location }),
    bullets: flags.bullets ?? [],
    ...(flags.techStack && { techStack: flags.techStack.split(',').map(t => t.trim()) }),
  };
}

// Similar for other add commands: buildSkillsFromFlags, buildEducationFromFlags, etc.

function normalizeUrl(url: string): string {
  if (url.startsWith('http://') || url.startsWith('https://')) return url;
  return `https://${url}`;
}
```

```typescript
import { type WizardState, createInitialState } from '../state.js';
import { validateWizardInput, WizardInitInputSchema, type StarBullet } from './schemas.js';
import { buildContactFromFlags, type InitFlagOptions } from './flag-collector.js';
import { exitWithError, EXIT_CODES } from './output-formatter.js';

/**
 * Detect conflicts between JSON data and CLI flags.
 * Per CONTEXT.md: error if both provide same field with different values.
 */
export function detectConflicts(
  jsonData: Record<string, unknown> | undefined,
  flags: Record<string, unknown>
): string[] {
  const conflicts: string[] = [];
  if (!jsonData) return conflicts;

  for (const [key, flagValue] of Object.entries(flags)) {
    if (flagValue !== undefined && jsonData[key] !== undefined) {
      // Compare JSON stringified values for deep comparison
      if (JSON.stringify(flagValue) !== JSON.stringify(jsonData[key])) {
        conflicts.push(`${key}: flag="${JSON.stringify(flagValue)}" vs json="${JSON.stringify(jsonData[key])}"`);
      }
    }
  }
  return conflicts;
}

/**
 * Convert STAR object to bullet string.
 */
function starToBullet(star: StarBullet): string {
  const parts: string[] = [];
  if (star.situation) parts.push(star.situation);
  parts.push(star.action);
  if (star.result) parts.push(star.result);
  return parts.join(' ');
}

/**
 * Process bullets that may be strings or STAR objects.
 */
function processBullets(bullets: Array<string | StarBullet>): string[] {
  return bullets.map(b => typeof b === 'string' ? b : starToBullet(b));
}

/**
 * Build wizard state from JSON input and/or flags.
 *
 * Priority per CONTEXT.md:
 * 1. Detect conflicts (error if found)
 * 2. JSON takes precedence for complex data (arrays, objects)
 * 3. Flags can supplement JSON data
 */
export function buildWizardState(
  jsonData: unknown | undefined,
  flags: InitFlagOptions,
  jsonMode: boolean
): WizardState {
  // Check for conflicts first
  if (jsonData && typeof jsonData === 'object') {
    const conflicts = detectConflicts(
      jsonData as Record<string, unknown>,
      flags as unknown as Record<string, unknown>
    );
    if (conflicts.length > 0) {
      exitWithError(
        'Conflicting values between flags and JSON input',
        conflicts,
        jsonMode,
        'VALIDATION_ERROR'
      );
    }
  }

  // Start with initial state
  const state = createInitialState('quick'); // Non-interactive uses quick mode

  // Apply JSON data if present
  if (jsonData) {
    const validated = validateWizardInput(WizardInitInputSchema, jsonData, jsonMode);

    state.contact = validated.contact;

    if (validated.experience) {
      state.experience = validated.experience.map(exp => ({
        ...exp,
        bullets: processBullets(exp.bullets),
      }));
    }

    if (validated.education) state.education = validated.education;
    if (validated.skills) state.skills = validated.skills;
    if (validated.projects) state.projects = validated.projects;
    if (validated.certifications) state.certifications = validated.certifications;
  }

  // Apply flags (only if JSON didn't provide the data)
  const flagState = buildContactFromFlags(flags);
  if (!state.contact && flagState.contact) {
    state.contact = flagState.contact;
  }

  return state;
}
```
