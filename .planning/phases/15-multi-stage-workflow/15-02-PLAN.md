1. `stages/analyze.ts`:
```typescript
import { generateText, Output } from 'ai';
import { renderPrompt } from '../../prompts/index.ts';
import { AnalyzeOutputSchema, type AnalyzeOutput } from '../schemas/index.ts';
import type { AIProvider } from '../../providers/types.ts';
import type { CVData } from '@gottz/cv-core';

interface AnalyzeStageInput {
  cv: CVData;
  locale: string;
  provider: AIProvider;
}

export async function runAnalyzeStage(input: AnalyzeStageInput): Promise<AnalyzeOutput> {
  const prompt = renderPrompt('analyze', {
    cv: input.cv,
    locale: input.locale,
  });

  const { output } = await generateText({
    model: input.provider.model,
    system: 'You are an expert CV consultant analyzing a CV for structure, content gaps, and improvement opportunities. Provide detailed, actionable feedback.',
    prompt,
    output: Output.object({
      schema: AnalyzeOutputSchema,
    }),
  });

  if (!output) {
    throw new Error('Analyze stage: LLM returned no structured output');
  }

  return output;
}
```

```typescript
import { generateText, Output } from 'ai';
import { renderPrompt } from '../../prompts/index.ts';
import { ImproveOutputSchema, type ImproveOutput, type AnalyzeOutput } from '../schemas/index.ts';
import type { AIProvider } from '../../providers/types.ts';
import type { CVData } from '@gottz/cv-core';

interface ImproveStageInput {
  cv: CVData;
  locale: string;
  provider: AIProvider;
  analyzeResult: AnalyzeOutput;
}

export async function runImproveStage(input: ImproveStageInput): Promise<ImproveOutput> {
  const prompt = renderPrompt('improve', {
    cv: input.cv,
    locale: input.locale,
    previousAnalysis: JSON.stringify(input.analyzeResult, null, 2),
  });

  const { output } = await generateText({
    model: input.provider.model,
    system: 'You are an expert CV writer. Improve achievement bullets using the STAR method (Situation, Task, Action, Result). Focus on quantifiable metrics and strong action verbs.',
    prompt,
    output: Output.object({
      schema: ImproveOutputSchema,
    }),
  });

  if (!output) {
    throw new Error('Improve stage: LLM returned no structured output');
  }

  return output;
}
```

1. `commands/ai/analyze.ts`:
```typescript
import ora from 'ora';
import pc from 'picocolors';
import { loadAIConfig, createProvider } from '../../ai/index.ts';
import {
  loadWorkflowState,
  saveWorkflowState,
  createInitialState,
  getStatePath
} from '../../ai/workflow/index.ts';
import { runAnalyzeStage } from '../../ai/workflow/stages/index.ts';
import { loadCV } from '../../lib/cv-loader.ts'; // Use existing CV loader

interface AnalyzeOptions {
  provider?: string;
  force?: boolean;
  quiet?: boolean;
  json?: boolean;
}

export async function analyzeAction(
  name: string | undefined,
  options: AnalyzeOptions
): Promise<void> {
  if (!name) {
    console.error(pc.red('Error: Person name is required'));
    console.error('Usage: cvgen ai analyze <name>');
    process.exit(1);
  }

  const personDir = `./people/${name}`;
  const cwd = process.cwd();

  // Load or create workflow state
  let state = await loadWorkflowState(personDir);

  if (state?.stageResults.analyze && !options.force) {
    if (!options.quiet) {
      console.log(pc.yellow('Analyze stage already complete. Use --force to re-run.'));
      console.log(`Status: ${pc.green('analyzed')}`);
    }
    if (options.json) {
      console.log(JSON.stringify(state.stageResults.analyze, null, 2));
    }
    return;
  }

  // Load CV data
  const { cv, locale, cvPath } = await loadCVData(personDir);

  // Create initial state if needed
  if (!state) {
    state = createInitialState(cvPath, locale);
  }

  // Get AI provider
  const config = await loadAIConfig(cwd);
  const provider = createProvider(config, options.provider as any);

  // Run analysis with spinner
  const spinner = ora('Analyzing CV...').start();

  try {
    const result = await runAnalyzeStage({
      cv,
      locale,
      provider,
    });

    // Update and save state
    state = {
      ...state,
      currentStage: 'analyzed',
      completedAt: { ...state.completedAt, analyzed: new Date().toISOString() },
      stageResults: { ...state.stageResults, analyze: result },
      lastUpdated: new Date().toISOString(),
    };
    await saveWorkflowState(personDir, state);

    spinner.succeed('Analysis complete');

    if (!options.quiet) {
      displayAnalyzeResult(result);
    }

    if (options.json) {
      console.log(JSON.stringify(result, null, 2));
    }

    if (!options.quiet && !options.json) {
      console.log(`\nNext: ${pc.cyan(`cvgen ai improve ${name}`)}`);
    }
  } catch (error) {
    spinner.fail('Analysis failed');
    console.error(pc.red(error instanceof Error ? error.message : 'Unknown error'));
    process.exit(1);
  }
}

function displayAnalyzeResult(result: AnalyzeOutput): void {
  console.log('\n' + pc.bold('Sections:'));
  for (const section of result.sections) {
    const statusColor = section.status === 'strong' ? pc.green :
                       section.status === 'needs_improvement' ? pc.yellow : pc.red;
    console.log(`  ${section.name}: ${statusColor(section.status)} (${section.bulletCount} bullets)`);
    if (section.issues.length > 0) {
      for (const issue of section.issues) {
        console.log(`    - ${pc.dim(issue)}`);
      }
    }
  }

  if (result.gaps.length > 0) {
    console.log('\n' + pc.bold('Gaps:'));
    for (const gap of result.gaps) {
      console.log(`  - ${gap}`);
    }
  }

  if (result.priorities.length > 0) {
    console.log('\n' + pc.bold('Priorities:'));
    for (const priority of result.priorities) {
      const impactColor = priority.impact === 'high' ? pc.red :
                         priority.impact === 'medium' ? pc.yellow : pc.dim;
      console.log(`  [${impactColor(priority.impact.toUpperCase())}] ${priority.section}: ${priority.issue}`);
    }
  }
}
```
