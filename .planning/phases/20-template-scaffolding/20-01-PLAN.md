<feature>
  <name>Template Copy with Overwrite Protection</name>
  <files>packages/cli/src/template/copier.ts, packages/cli/src/template/__tests__/copier.test.ts</files>
  <behavior>
    copyTemplate(sourceId, targetId, templatesDir) copies a template directory.

    Cases:
    - copyTemplate("modern", "my-custom", dir) -> creates dir/my-custom/ with all files from dir/modern/
    - config.json in target has name updated: "modern" -> "My Custom"
    - config.json in target has private: true removed if present
    - copyTemplate("modern", "existing", dir) where existing exists -> throws error
    - formatTemplateName("my-custom") -> "My Custom"
    - formatTemplateName("my-test-template") -> "My Test Template"
  </behavior>
  <implementation>
    Use fs.cp() with { recursive: true, errorOnExist: true, force: false }.
    After copy, read config.json, update name using formatTemplateName, delete private field, write back.
    formatTemplateName splits on hyphens and capitalizes each word.
  </implementation>
</feature>

<feature>
  <name>Template Validation</name>
  <files>packages/cli/src/template/validator.ts, packages/cli/src/template/__tests__/validator.test.ts</files>
  <behavior>
    validateTemplate(templateId, templatesDir) returns ValidationResult with valid, errors, warnings.

    Cases:
    - Valid template with all files -> { valid: true, errors: [], warnings: [] }
    - Missing config.json -> { valid: false, errors: ["Missing required file: config.json"], warnings: [] }
    - Missing template.njk -> { valid: false, errors: ["Missing required file: template.njk"], warnings: [] }
    - Missing styles.css -> { valid: false, errors: ["Missing required file: styles.css"], warnings: [] }
    - Invalid JSON in config.json -> { valid: false, errors: ["config.json: invalid JSON - ..."], warnings: [] }
    - config.json missing name field -> { valid: false, errors: ["config.json: missing \"name\" field"], warnings: [] }
    - config.json missing description -> { valid: false, errors: ["config.json: missing \"description\" field"], warnings: [] }
    - Invalid hex color in style.accentColor -> warning (not error, template still works)
    - Template directory doesn't exist -> { valid: false, errors: ["Template directory not found: {id}"], warnings: [] }
  </behavior>
  <implementation>
    Check fs.access for required files. Parse config.json with try/catch for JSON errors.
    Validate required fields. Use isValidHexColor from @gottz/cv-templates for color warnings.
    Return structured ValidationResult.
  </implementation>
</feature>
