    (function() {
      const toggle = document.querySelector('.theme-toggle');
      const html = document.documentElement;
      const STORAGE_KEY = 'cv-theme';
      const THEMES = ['system', 'light', 'dark'];

      function getStoredTheme() {
        try { return localStorage.getItem(STORAGE_KEY); } catch { return null; }
      }

      function setStoredTheme(theme) {
        try {
          if (theme === 'system') localStorage.removeItem(STORAGE_KEY);
          else localStorage.setItem(STORAGE_KEY, theme);
        } catch {}
      }

      function applyTheme(theme) {
        html.setAttribute('data-theme', theme);
        toggle.setAttribute('aria-label', {
          system: 'Current: system preference. Click to switch to light mode.',
          light: 'Current: light mode. Click to switch to dark mode.',
          dark: 'Current: dark mode. Click to switch to system preference.'
        }[theme] || 'Toggle theme');
      }

      function cycleTheme() {
        const current = html.getAttribute('data-theme') || 'system';
        const next = THEMES[(THEMES.indexOf(current) + 1) % THEMES.length];
        applyTheme(next);
        setStoredTheme(next);
      }

      const stored = getStoredTheme();
      applyTheme(stored && THEMES.includes(stored) ? stored : 'system');
      toggle.addEventListener('click', cycleTheme);
    })();
  </script>
</body>
</html>
