/**
 * Standalone summary generator - not part of workflow state.
 * Reuses SummarizeOutputSchema from Phase 15.
 */

import { generateText, Output } from 'ai';
import type { CVData } from '@gottz/cv-core';
import type { AIProvider } from '../providers/types.ts';
import { SummarizeOutputSchema, type SummarizeOutput } from '../workflow/schemas/summarize-output.ts';
import { renderPrompt } from '../prompts/index.ts';
import { withRetry } from '../utils/retry.ts';

export interface SummaryOptions {
  targetRole?: string;
}

export async function generateSummary(
  cv: CVData,
  locale: string,
  provider: AIProvider,
  options: SummaryOptions = {}
): Promise<SummarizeOutput> {
  const prompt = await renderPrompt('summary-gen', {
    cv_content: JSON.stringify(cv, null, 2),
    locale,
    target_role: options.targetRole,
  });

  const result = await withRetry(async () => {
    const { output } = await generateText({
      model: provider.model,
      prompt,
      output: Output.object({ schema: SummarizeOutputSchema }),
    });
    return output;
  });

  return result;
}
```

/**
 * AI summary subcommand - generate professional summary (AI-07).
 * Standalone generator, not part of workflow state.
 * Note: Distinct from 'summarize' workflow stage command.
 */

import path from 'node:path';
import { writeFile } from 'node:fs/promises';
import ora from 'ora';
import pc from 'picocolors';
import { createProvider, loadAIConfig } from '../../ai/index.ts';
import { loadCVForStage } from '../../ai/workflow/index.ts';
import { generateSummary } from '../../ai/generators/index.ts';
import { formatSummaryAsCvMd } from '../../ai/display/index.ts';

export interface SummaryOptions {
  targetRole?: string;
  provider?: string;
  output?: string;
  quiet?: boolean;
  json?: boolean;
}

export async function summaryAction(
  name: string | undefined,
  options: SummaryOptions
): Promise<void> {
  if (!name) {
    console.error(pc.red('Error: Person name is required'));
    console.error('Usage: cvgen ai summary <name>');
    process.exit(1);
  }

  const cwd = process.cwd();
  const personDir = path.join(cwd, 'people', name);

  // Load CV
  const { cv, locale } = await loadCVForStage(personDir);

  // Get AI provider
  const config = await loadAIConfig(cwd);
  const provider = createProvider(config, options.provider as any);

  // Generate with spinner
  const spinner = options.quiet ? null : ora('Generating summary...').start();

  try {
    const result = await generateSummary(cv, locale, provider, {
      targetRole: options.targetRole,
    });

    spinner?.succeed('Summary generated');

    // Format output
    if (options.json) {
      console.log(JSON.stringify(result, null, 2));
    } else {
      const formatted = formatSummaryAsCvMd(result.primary, locale);

      if (options.output) {
        await writeFile(options.output, formatted, 'utf-8');
        console.log(pc.green(`Written to ${options.output}`));
      } else {
        console.log('\n' + formatted);
      }

      // Show alternative and key points
      if (!options.quiet) {
        console.log(`\n${pc.bold('Alternative:')}`);
        console.log(pc.dim(result.alternative));

        console.log(`\n${pc.bold('Key Points:')}`);
        for (const point of result.keyPoints) {
          console.log(`  - ${point}`);
        }

        console.log(`\n${pc.bold('Target Roles:')}`);
        console.log(`  ${result.targetRoles.join(', ')}`);
      }
    }
  } catch (error) {
    spinner?.fail('Summary generation failed');
    console.error(pc.red(error instanceof Error ? error.message : 'Unknown error'));
    process.exit(1);
  }
}
```
