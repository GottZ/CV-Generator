Create prompt template at `packages/cli/src/ai/prompts/templates/improve-standalone.md`:

```markdown
# Improve CV Content

Analyze the CV and suggest improvements for weak content.

## CV Content
{{ cv_content }}

## Instructions

For each section of the CV:
1. Identify weak bullets (lacking impact, metrics, or specificity)
2. Generate improved versions using STAR method
3. Assign priority based on impact potential
4. Provide specific feedback on why each bullet needs improvement

## Priority Levels
- High: Critical issues - vague responsibilities, no outcomes, passive voice
- Medium: Good but improvable - could add metrics or specificity
- Low: Minor polish - style or wording tweaks

## Weak Bullet Indicators
- Lacks quantification ("improved efficiency" vs "improved efficiency by 40%")
- Missing outcome ("worked on project" vs "delivered project reducing costs by $50K")
- Too generic ("responsible for" vs "led team of 5 engineers")
- Passive voice ("was assigned" vs "took ownership")

## Output Requirements
For each suggestion:
- Original text
- Improved text
- Reasoning (specific feedback)
- Priority level
- Metrics extracted or suggested

CRITICAL: Only use metrics from the CV. If suggesting new metrics, clearly mark as "estimated" or "example".
```

/**
 * Standalone improvement generator with priority grouping.
 * Separate from workflow improve stage for richer standalone use.
 */

import { generateText, Output } from 'ai';
import { z } from 'zod';
import type { CVData } from '@gottz/cv-core';
import type { AIProvider } from '../providers/types.ts';
import { renderPrompt } from '../prompts/index.ts';
import { withRetry } from '../utils/retry.ts';

const ImprovementSchema = z.object({
  section: z.string().describe('CV section containing the bullet'),
  original: z.string(),
  improved: z.string(),
  reasoning: z.string().describe('Specific feedback on why this needs improvement'),
  priority: z.enum(['high', 'medium', 'low']),
  metrics: z.array(z.string()).optional(),
  weaknessType: z.string().optional()
    .describe('lacks_quantification, missing_outcome, too_generic, passive_voice'),
});

export const ImprovementsOutputSchema = z.object({
  improvements: z.array(ImprovementSchema),
  summary: z.object({
    total: z.number(),
    high: z.number(),
    medium: z.number(),
    low: z.number(),
  }),
  overallAssessment: z.string()
    .describe('Brief assessment of CV content quality'),
});

export type ImprovementsOutput = z.infer<typeof ImprovementsOutputSchema>;
export type Improvement = z.infer<typeof ImprovementSchema>;

export async function generateImprovements(
  cv: CVData,
  locale: string,
  provider: AIProvider
): Promise<ImprovementsOutput> {
  const prompt = await renderPrompt('improve-standalone', {
    cv_content: JSON.stringify(cv, null, 2),
    locale,
  });

  const result = await withRetry(async () => {
    const { output } = await generateText({
      model: provider.model,
      prompt,
      output: Output.object({ schema: ImprovementsOutputSchema }),
    });
    return output;
  });

  return result;
}

/**
 * AI improve subcommand - standalone version with rich diff display (AI-09).
 * Enhances the workflow improve command with better visualization.
 */

import path from 'node:path';
import ora from 'ora';
import pc from 'picocolors';
import { createProvider, loadAIConfig } from '../../ai/index.ts';
import { loadCVForStage } from '../../ai/workflow/index.ts';
import { generateImprovements, type ImprovementsOutput, type Improvement } from '../../ai/generators/improve-standalone.ts';
import { displayComparison } from '../../ai/display/diff-display.ts';
import { formatPriority } from '../../ai/display/quality-labels.ts';

export interface ImproveStandaloneOptions {
  provider?: string;
  quiet?: boolean;
  json?: boolean;
}

export async function improveStandaloneAction(
  name: string | undefined,
  options: ImproveStandaloneOptions
): Promise<void> {
  if (!name) {
    console.error(pc.red('Error: Person name is required'));
    console.error('Usage: cvgen ai improve <name>');
    process.exit(1);
  }

  const cwd = process.cwd();
  const personDir = path.join(cwd, 'people', name);

  // Load CV
  const { cv, locale } = await loadCVForStage(personDir);

  // Get AI provider
  const config = await loadAIConfig(cwd);
  const provider = createProvider(config, options.provider as any);

  // Generate with spinner
  const spinner = options.quiet ? null : ora('Analyzing improvements...').start();

  try {
    const result = await generateImprovements(cv, locale, provider);

    spinner?.succeed('Analysis complete');

    if (options.json) {
      console.log(JSON.stringify(result, null, 2));
      return;
    }

    // Display summary
    console.log(`\n${pc.bold('Improvement Summary:')}`);
    console.log(`  Total suggestions: ${result.summary.total}`);
    console.log(`  ${pc.red('High priority:')} ${result.summary.high}`);
    console.log(`  ${pc.yellow('Medium priority:')} ${result.summary.medium}`);
    console.log(`  ${pc.dim('Low priority:')} ${result.summary.low}`);
    console.log(`\n${result.overallAssessment}`);

    // Group by priority
    const byPriority = {
      high: result.improvements.filter(i => i.priority === 'high'),
      medium: result.improvements.filter(i => i.priority === 'medium'),
      low: result.improvements.filter(i => i.priority === 'low'),
    };

    // Display high priority first
    for (const priority of ['high', 'medium', 'low'] as const) {
      const items = byPriority[priority];
      if (items.length === 0) continue;

      console.log(`\n${pc.bold(`${formatPriority(priority)} Priority Improvements:`)}`);

      for (const item of items) {
        console.log(`\n${pc.dim(`[${item.section}]`)} ${item.weaknessType ? pc.dim(`(${item.weaknessType})`) : ''}`);

        // Use diff display
        const diff = displayComparison(item.original, item.improved);
        console.log(diff);

        console.log(`${pc.dim('Reasoning:')} ${item.reasoning}`);

        if (item.metrics && item.metrics.length > 0) {
          console.log(`${pc.cyan('Metrics:')} ${item.metrics.join(', ')}`);
        }
      }
    }

    // Footer with next steps
    console.log(`\n${pc.dim('To accept changes, copy the improved text to your cv.md file.')}`);

  } catch (error) {
    spinner?.fail('Analysis failed');
    console.error(pc.red(error instanceof Error ? error.message : 'Unknown error'));
    process.exit(1);
  }
}
