1. Add utility function to convert text with newlines to TextRun array:
```typescript
/**
 * Convert text with newlines to array of TextRuns.
 * Single \n becomes a line break within the paragraph.
 * Use for content that should stay in one paragraph but have line breaks.
 */
function textWithBreaks(text: string, options?: Partial<IRunOptions>): TextRun[] {
  const lines = text.split('\n');
  const runs: TextRun[] = [];

  for (let i = 0; i < lines.length; i++) {
    if (i > 0) {
      // Add line break before this line
      runs.push(new TextRun({ ...options, break: 1 }));
    }
    if (lines[i].trim()) {
      runs.push(new TextRun({ text: lines[i], ...options }));
    }
  }

  return runs;
}
```
