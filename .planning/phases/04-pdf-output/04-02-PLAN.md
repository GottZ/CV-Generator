    export interface PdfMetadata {
      title: string;      // "Name - CV"

    1. Load PDF: `const pdfDoc = await PDFDocument.load(pdfBuffer)`
    2. Set metadata:

       - `pdfDoc.setAuthor(metadata.author)`
       - `pdfDoc.setSubject(metadata.subject ?? 'Curriculum Vitae')`
       - `pdfDoc.setCreator(metadata.creator ?? 'CV Generator')`
       - `pdfDoc.setKeywords(metadata.keywords ?? [])` if provided
       - `pdfDoc.setCreationDate(new Date())`
       - `pdfDoc.setModificationDate(new Date())`
    3. Save and return: `Buffer.from(await pdfDoc.save())`

    Import: `import { PDFDocument } from 'pdf-lib'`

<task type="auto">
  <name>Task 2: Create pdf-bookmarks.ts for section navigation</name>
  <files>packages/cli/src/lib/pdf-bookmarks.ts</files>
  <action>
    Create PDF bookmark module using @lillallol/outline-pdf per RESEARCH.md Pattern 4.

    Interface:
    ```typescript
    export interface SectionInfo {
      title: string;  // Section header (e.g., "Experience", "Education")
      page: number;   // 1-indexed page number
    }
    ```

    Function `addPdfBookmarks(pdfBuffer: Buffer, sections: SectionInfo[]): Promise<Buffer>`:
    1. Import and initialize:
       ```typescript
       import * as pdfLib from 'pdf-lib';
       import { outlinePdfFactory } from '@lillallol/outline-pdf';
       const outlinePdf = outlinePdfFactory(pdfLib);
       ```
    2. Build outline string format: "pageNum||title"
       ```typescript
       const outlineString = sections
         .map(s => `${s.page}||${s.title}`)
         .join('\n');
       ```
    3. Apply outline:
       ```typescript
       const pdfDoc = await outlinePdf({
         pdf: pdfBuffer,
         outline: outlineString,
       });
       ```
    4. Save and return: `Buffer.from(await pdfDoc.save())`

    Edge case: If sections array is empty, return original buffer unchanged.

    Note: Page numbers are determined externally. For Phase 4, we'll assume all CVs fit on 1-2 pages and sections start on page 1. More sophisticated page detection can be added later if needed.

    Default sections for bookmark generation (i18n aware):
    ```typescript
    export function getDefaultSections(locale: string): SectionInfo[] {
      if (locale === 'de') {
        return [
          { title: 'Berufserfahrung', page: 1 },
          { title: 'Ausbildung', page: 1 },
          { title: 'Kenntnisse', page: 1 },
        ];
      }
      return [
        { title: 'Experience', page: 1 },
        { title: 'Education', page: 1 },
        { title: 'Skills', page: 1 },
      ];
    }
    ```
  </action>

    - File exports addPdfBookmarks, SectionInfo, getDefaultSections
    - Uses outlinePdfFactory from @lillallol/outline-pdf
    - Handles empty sections array gracefully
    - Supports both en and de locales
  </verify>
