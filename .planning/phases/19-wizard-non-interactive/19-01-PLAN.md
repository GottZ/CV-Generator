```typescript
export interface WizardModeOptions {
  noInput?: boolean;        // --no-input flag
  forceInteractive?: boolean; // --force-interactive flag
  jsonInput?: string;       // --json-input file path or '-' for stdin
  dryRun?: boolean;         // --dry-run flag
  enhance?: boolean;        // --enhance flag
  json?: boolean;           // --json flag for output format
}

export type WizardExecutionMode = 'interactive' | 'non-interactive';

export function detectMode(options: WizardModeOptions): WizardExecutionMode;
export function ensureNonInteractiveRequirements(
  hasRequiredInput: boolean,
  jsonOutput: boolean
): void;
```

Implementation logic per CONTEXT.md:
1. If --force-interactive: return 'interactive' (escape hatch)
2. If --no-input: return 'non-interactive'
3. If process.stdin.isTTY is false: return 'non-interactive' (auto-switch)
4. Otherwise: return 'interactive'

The ensureNonInteractiveRequirements function should:
- Check if non-interactive mode has required input (flags or --json-input)
- If not, show helpful error message and exit with code 2
- Error format respects --json flag (JSON error if --json, human-readable otherwise)
