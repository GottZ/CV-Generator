```typescript
/**
 * Zod schema for keyword analysis output.
 * Source: RESEARCH.md "Keyword Analysis Output Schema"
 */

import { z } from 'zod';

const KeywordEntrySchema = z.object({
  keyword: z.string(),
  found: z.boolean(),
  suggestedPlacement: z.string().optional()
    .describe('Which CV section this keyword should appear in'),
  rewrittenContent: z.string().optional()
    .describe('Suggested rewrite integrating the keyword'),
  context: z.string().optional()
    .describe('2-3 lines of surrounding context'),
});

const SectionSuggestionsSchema = z.object({
  section: z.string().describe('CV section: skills, experience, summary'),
  suggestions: z.array(z.object({
    keyword: z.string(),
    currentText: z.string().optional(),
    rewrittenText: z.string(),
    context: z.string().describe('2-3 lines surrounding context'),
  })),
});

export const KeywordsOutputSchema = z.object({
  score: z.number().min(0).max(100)
    .describe('Overall keyword coverage percentage'),

  byCategory: z.object({
    required: z.object({
      total: z.number(),
      matched: z.number(),
      keywords: z.array(KeywordEntrySchema),
    }),
    preferred: z.object({
      total: z.number(),
      matched: z.number(),
      keywords: z.array(KeywordEntrySchema),
    }),
  }),

  bySection: z.array(SectionSuggestionsSchema),

  highCoverage: z.boolean()
    .describe('True if score > 90%'),

  summary: z.string()
    .describe('Brief summary of keyword coverage status'),
});

export type KeywordsOutput = z.infer<typeof KeywordsOutputSchema>;
```

```markdown
# Analyze ATS Keywords

Analyze the CV against this job posting to identify missing keywords and suggest placements.

## CV Content
{{ cv_content }}

## Job Description
{{ job_description }}

## Instructions

1. Extract required and preferred keywords from the job description
2. Check which keywords are present in the CV ({{ match_mode }} matching)
3. For missing keywords, suggest where to add them in the CV
4. Provide rewritten content integrating missing keywords naturally
5. Calculate overall keyword coverage percentage

## Keyword Categories
- Required: Must-have skills, qualifications, or terms
- Preferred: Nice-to-have skills or bonus qualifications

## Placement Rules
- Use EXACT phrasing from job posting (e.g., "Machine Learning" not "ML")
- Suggest natural integration, not keyword stuffing
- Include 2-3 lines of surrounding context for each suggestion

{% if exact_matching %}
Use exact case-insensitive matching only.
{% else %}
Use fuzzy matching for abbreviations and variations (JS/JavaScript, K8s/Kubernetes).
{% endif %}

## Quality Guidelines
- High coverage (>90%): Celebrate, suggest minor additions
- Medium coverage (50-89%): Prioritize missing required keywords
- Low coverage (<50%): Focus on critical gaps first
```

```typescript
/**
 * Keyword analysis generator - uses LLM for extraction plus local matching.
 */

import { generateText, Output } from 'ai';
import type { CVData } from '@gottz/cv-core';
import type { AIProvider } from '../providers/types.ts';
import { KeywordsOutputSchema, type KeywordsOutput } from '../workflow/schemas/keywords-output.ts';
import { renderPrompt } from '../prompts/index.ts';
import { withRetry } from '../utils/retry.ts';
import { matchKeywords } from '../utils/keyword-matcher.ts';

export interface KeywordsOptions {
  exact?: boolean;
}

export async function analyzeKeywords(
  cv: CVData,
  locale: string,
  provider: AIProvider,
  jobDescription: string,
  options: KeywordsOptions = {}
): Promise<KeywordsOutput> {
  const prompt = await renderPrompt('keywords', {
    cv_content: JSON.stringify(cv, null, 2),
    job_description: jobDescription,
    locale,
    match_mode: options.exact ? 'exact' : 'fuzzy',
    exact_matching: options.exact,
  });

  const result = await withRetry(async () => {
    const { output } = await generateText({
      model: provider.model,
      prompt,
      output: Output.object({ schema: KeywordsOutputSchema }),
    });
    return output;
  });

  return result;
}
```

```typescript
/**
 * AI keywords subcommand - ATS keyword optimization (AI-08).
 */

import path from 'node:path';
import ora from 'ora';
import pc from 'picocolors';
import { createProvider, loadAIConfig } from '../../ai/index.ts';
import { loadCVForStage } from '../../ai/workflow/index.ts';
import { analyzeKeywords } from '../../ai/generators/index.ts';
import { loadJobDescription } from '../../ai/utils/index.ts';
import { formatKeywordScore } from '../../ai/display/quality-labels.ts';

export interface KeywordsOptions {
  job?: string;
  exact?: boolean;
  provider?: string;
  quiet?: boolean;
  json?: boolean;
}

export async function keywordsAction(
  name: string | undefined,
  options: KeywordsOptions
): Promise<void> {
  if (!name) {
    console.error(pc.red('Error: Person name is required'));
    process.exit(1);
  }

  if (!options.job) {
    console.error(pc.red('Error: Job description is required'));
    console.error('Usage: cvgen ai keywords <name> --job <file>');
    process.exit(1);
  }

  const cwd = process.cwd();
  const personDir = path.join(cwd, 'people', name);

  // Load CV and job description
  const { cv, locale } = await loadCVForStage(personDir);
  const jobSource = await loadJobDescription(options.job);

  // Get AI provider
  const config = await loadAIConfig(cwd);
  const provider = createProvider(config, options.provider as any);

  // Analyze with spinner
  const spinner = options.quiet ? null : ora('Analyzing keywords...').start();

  try {
    const result = await analyzeKeywords(cv, locale, provider, jobSource.content, {
      exact: options.exact,
    });

    spinner?.succeed('Keyword analysis complete');

    if (options.json) {
      console.log(JSON.stringify(result, null, 2));
      return;
    }

    // Display score prominently
    console.log(`\n${pc.bold('Keyword Coverage:')} ${formatKeywordScore(result.score)}`);

    if (result.highCoverage) {
      console.log(pc.green('\nExcellent coverage! Your CV matches most job requirements.'));
    }

    // Display summary
    console.log(`\n${result.summary}`);

    // Required keywords
    console.log(`\n${pc.bold('Required Keywords:')} (${result.byCategory.required.matched}/${result.byCategory.required.total})`);
    for (const kw of result.byCategory.required.keywords) {
      const status = kw.found ? pc.green('Found') : pc.red('Missing');
      console.log(`  ${status} - ${kw.keyword}`);
      if (!kw.found && kw.suggestedPlacement) {
        console.log(`    ${pc.dim('Suggestion:')} Add to ${kw.suggestedPlacement}`);
      }
    }

    // Preferred keywords
    console.log(`\n${pc.bold('Preferred Keywords:')} (${result.byCategory.preferred.matched}/${result.byCategory.preferred.total})`);
    for (const kw of result.byCategory.preferred.keywords) {
      const status = kw.found ? pc.green('Found') : pc.yellow('Missing');
      console.log(`  ${status} - ${kw.keyword}`);
    }

    // Section suggestions
    if (result.bySection.length > 0) {
      console.log(`\n${pc.bold('Placement Suggestions:')}`);
      for (const section of result.bySection) {
        console.log(`\n  ${pc.cyan(section.section)}:`);
        for (const sug of section.suggestions) {
          console.log(`    Keyword: ${pc.bold(sug.keyword)}`);
          if (sug.currentText) {
            console.log(`    Current: ${pc.dim(sug.currentText)}`);
          }
          console.log(`    Rewrite: ${pc.green(sug.rewrittenText)}`);
        }
      }
    }
  } catch (error) {
    spinner?.fail('Keyword analysis failed');
    console.error(pc.red(error instanceof Error ? error.message : 'Unknown error'));
    process.exit(1);
  }
}
```
