Create the wizard orchestration module:

**runner.ts:**
- Import pc from picocolors
- Import all prompt collectors from ./prompts/index.ts
- Import { showMainMenu, selectMode } from ./menu.ts
- Import { createInitialState, createStateFromExisting, isMinimumViable } from ./state.ts
- Import { confirmSave } from ./summary.ts
- Import { writeWizardOutput } from ./markdown-writer.ts
- Import { ensureInteractiveMode } from '../ai/review/tty-check.ts'
- Import { parseCV } from '@gottz/cv-core'
- Import { readFile, access } from 'node:fs/promises'
- Import path from 'node:path'

- WizardOptions interface:
  ```typescript
  interface WizardOptions {
    personDir: string;  // Full path to person directory
    personName: string; // Directory name (e.g., "jane-doe")
    locale?: string;    // Default: 'en'
  }
  ```

- setupCleanExit(): void
  - Per RESEARCH.md: Handle ExitPromptError for clean Ctrl+C exit
  - Register process.on('uncaughtException') handler
  - Check error.name === 'ExitPromptError'
  - Log: pc.dim('\nWizard cancelled. No changes saved.')
  - Exit with code 130 (standard SIGINT)
  - Rethrow non-ExitPromptError errors

- runWizard(options: WizardOptions): Promise<void>
  - Call ensureInteractiveMode() first (WIZ-18 prep for Phase 19)
  - Call setupCleanExit()
  - Try to load existing CV if cv.md exists in personDir:
    - If exists: parse with parseCV, create state with createStateFromExisting
    - Log: pc.cyan('Editing existing CV for {personName}')
    - If not exists: select mode, create empty state
  - Main loop:
    - Call showMainMenu(state)
    - Switch on selection:
      - 'contact': state.contact = await collectContact(state.mode, state.contact)
      - 'experience': state.experience = await collectExperience(state.mode, state.experience)
      - 'education': state.education = await collectEducation(state.mode, state.education)
      - 'skills': state.skills = await collectSkills(state.mode, state.skills)
      - 'projects': state.projects = await collectProjects(state.mode, state.projects)
      - 'certifications': state.certifications = await collectCertifications(state.mode, state.certifications)
      - 'finish': break loop
      - 'cancel': log message and return
    - Update currentSection in state
  - Check isMinimumViable(state):
    - If false: warn and return to menu
  - Call confirmSave(state):
    - 'confirm': await writeWizardOutput(state, options.personDir, options.locale)
    - 'edit': continue to menu
    - 'cancel': log message and return

- runAddSection(section: string, options: WizardOptions): Promise<void>
  - Call ensureInteractiveMode()
  - Call setupCleanExit()
  - Load existing CV (required for add commands)
  - If no cv.md: error with "Use 'wizard init' to create a new CV first"
  - Select mode for this session
  - Call appropriate collector based on section
  - Show mini-summary of what was added
  - Confirm and write
